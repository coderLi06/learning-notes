public static ListNode mergeTwoLists(ListNode head1, ListNode head2){
    //是否有链表为空
    if(head1 == null || head2 == null){
        return head1 == null ? head2 : head1;
    }
    ListNode head = head1.val <= head2.val ? head1 : head2;//整个头节点
    ListNode cur1 = head.next;//头结点下一个
    ListNode cur2 = head == head1 ? head2 : head1;//另一个头结点
    ListNode pre = head;//已调好节点
    while(cur1 != null && cur2 != null){
		if(cur1.val <= cur2.val){
            pre.next = cur1;//指针指向最后一个已调好节点
            cur1 = cur1.next;
        }else{
            pre.next = cur2;
            cur2 = cur2.next;
        }
        pre = pre.next;//更新已调好节点位置
    }
    pre.next = cur1 == null ? cur2 : cur1;//指针指向调完一条链表的另一条位置
    return head;
}
