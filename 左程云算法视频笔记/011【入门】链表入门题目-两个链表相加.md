//力扣2.两数相加
public static ListNode addTwoNumbers(ListNode h1, ListNode h2){
    //ans指向头结点，cur指向答案链表的当前节点
    ListNode ans = null, cur = null;
    int carry = 0;
    for(int sum, val;//当前两数和，当前位结果
        h1 != null || h2 != null;//终止条件为都为空
        h1 = h1 == null ? null : h1.next,
        h2 = h2 == null ? null : h2.next){
        sum = (h1==null ? 0 : h1.val) + (h2 == null ? 0 : h2.val)+carry;
        val = sum % 10;
        carry = sum / 10;
        if(ans == null){
            ans = new ListNode(val);
            cur = ans;
        }else{
            cur.next = new ListNode(val);
            cur = cur.next;
        }
    }
    if(carry == 1) cur.next = new ListNode(1);//.next说明指向关系
    return ans;
}
