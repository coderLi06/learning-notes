```java
import java.io.*;

/**
 * 归并分治适用范围：
 * 条件一：一个大范围的问题可以分解为：左范围内单独结果+右范围内单独结果+跨越左右产生的结果
 * 条件二：计算跨越左右产生的答案时，如果加上左右有序的设定，会不会获得计算的便利
 * 如果都成立，很可能可以使用归并分治
 * 
 */
// ACM风格IO流模板

public class Merge {
    public static int MAXN = 100001;
    public static int[] arr = new int[MAXN];
    public static int[] help = new int[MAXN];
    public static int n;// 代表数组的真实长度

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StreamTokenizer in = new StreamTokenizer(br);
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
        while (in.nextToken() != StreamTokenizer.TT_EOF) {
            n = (int) in.nval;
            for (int i = 0; i < n; i++) {
                in.nextToken();
                arr[i] = (int) in.nval;
            }
            out.println(smallSum(0, n - 1));
        }
        out.flush();
        out.close();
    }

    /*
     * 小和：一个数之前的，所有比这个数小的数，之和
     * 结果比较大，用long类型，特别注意，笔试常见坑
     * 返回arr[l...r]范围小和的累加和，同时把arr变有序
     */
    public static long smallSum(int l, int r) {
        if (l == r)
            return 0;
        int m = (l + r) / 2;
        return smallSum(l, m) + smallSum(m + 1, r) + merge(l, m, r);
    }

    public static long merge(int l, int m, int r) {
        // 第一个功能：计算小和
        long ans = 0;// 记录累积结果
        for (int i = l, j = m + 1, sum = 0; j <= r; j++) {
            while (i <= m && arr[i] <= arr[j]) {
                sum += arr[i++];// sum左边数组中小于右边arr[j]的累积值
            }
            ans += sum;// j每更新一次，ans加一个sum
        }
        // 第二个功能：把arr[l...r]变有序
        int a = l, b = m + 1, k = l;
        while (a <= m && b <= r) {
            help[k++] = arr[a++] <= arr[b++] ? arr[a++] : arr[b++];
        }
        while (a <= m)
            help[k++] = arr[a++];
        while (b <= r)
            help[k++] = arr[b++];
        for (int i = l; i <= r; i++) {
            arr[i] = help[i];
        }
        return ans;

    }

}
```

## leetcode493.翻转对
给定一个数组 nums ，如果 i < j 且 nums[i] > 2*nums[j] 我们就将 (i, j) 称作一个重要翻转对。

你需要返回给定数组中的重要翻转对的数量
```java
class Solution {
    public static int MAXN = 50001;
    public static int[] help = new int[MAXN];
    public int reversePairs(int[] nums) {
        return counts(nums, 0, nums.length-1);
    }

    public static int counts(int[] arr, int l, int r){
        if(l == r) return 0;
        int m = (l+r)/2;
        return counts(arr, l, m)+counts(arr,m+1,r)
        +merge(arr, l, m, r);
    }
    public static int merge(int[] arr, int l, int m, int r){
        //1.统计翻转对数量
        long ans = 0;
        for(int i=l,j=m+1; i<=m; i++){//j往左全部小于(不含j)
            while(j<=r && (long)arr[i]>(long)arr[j]*2){//注意数据溢出
                j++;
            }
            ans += j-m-1;
        }
        //2.正常归并排序
        int i=l, j=m+1, k=l;
        while(i<=m && j<=r) help[k++] = arr[i] <= arr[j] ? arr[i++] : arr[j++];
        while(i<=m) help[k++] = arr[i++];
        while(j<=r) help[k++] = arr[j++];
        for(int a = l; a<=r; a++) arr[a] = help[a];
        return ans;
    }
}
```
