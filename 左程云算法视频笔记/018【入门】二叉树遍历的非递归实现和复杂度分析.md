//非递归先序打印，借助栈
//最先压入的最后处理
public static List<Integer> preOrder(TreeNode head){
    List<Integer> res = new ArrayList<>();
    if(head != null){
        Stack<TreeNode> stack = new Stack<>();
        stack.push(head);//1.压入头节点

        while(!stack.isEmpty()){
            head = stack.pop();
            res.add(head.val);//打印头节点值
            if(head.right != null) stack.push(head.right);//2.先压右，保证左子树优先打印

            if(head.left != null) stack.push(head.left);//3.再压左
        }
    }
    return res;
}

//中序打印
public static List<Integer> inOrder(TreeNode head){
    List<Integer> res = new ArrayList<>();

    if(head != null){
        Stack<TreeNode> stack = new Stack<>();
        //处理到根节点时，栈内为空，但head=root.right，依旧执行循环
        while(!stack.isEmpty() || head != null){
            //先处理左树，再处理自己和右树
            if(head != null){//一直压入左子树
                stack.push(head);
                head = head.left;
            }else{
                head = stack.pop();
                res.add(head.val);//处理左树
                head = head.right;//处理右子树，回到上方继续优先处理左子树
            }
        }
    }
    return res;
}


/*
后序打印，两个栈
先序：中左右 -> 中右左 -> 左右中
调整先序顺序，从先压入右变成先压入左，再用栈反转
*/
public static List<Integer> postOrderTwoStacks(TreeNode head){
    List<Integer> res = new ArrayList<>();
    if(head != null){
        Stack<TreeNode> stack = new Stack<>();
        Stack<TreeNode> collect = new Stack<>();
        stack.push(head);
        while(!stack.isEmpty()){
            head = stack.pop();
            collect.push(head);//原本该打印处理的位置，改成收集，实现逆序
            if(head.left != null) stack.push(head.left);
            if(head.right != null) stack.push(head.right);
        }
        while(!collect.isEmpty()){
            res.add(collect.pop().val);//反转打印顺序，即为后序遍历
        }
    }
    return res;
}

//后序打印，一个栈
public static List<Integer> postOrderOneStack(TreeNode head){
    List<Integer> res = new ArrayList<>();
    //使用指针h标记当前已处理过的节点
    TreeNode h = head;
    if(h != null){
        Stack<TreeNode> stack = new Stack<>();
        stack.push(h);
        while(!stack.isEmpty()){
            //使用cur标记当前节点当前节点
            TreeNode cur = stack.peek();

            //左右子树都不等于h，说明都没处理，先处理左
            if(cur.left != null && h != cur.left && h != cur.right){
                stack.push(cur.left);
            }//左子树没有或者已经处理，且右子树没处理，处理右
            else if(cur.right != null && h != cur.right){
                stack.push(cur.right);
            }//都处理过，处理自己
            else{
                res.add(cur.val);
                h = stack.pop();//弹出当前节点，继续处理下一个节点
            }
        }
    }
    return res;
}
