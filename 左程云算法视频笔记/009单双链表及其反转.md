//力扣206反转链表（单链表）
public static ListNode reverseList(ListNode head){
    ListNode pre = null;
    ListNode next = null;
    //需要更新的元素：head, head.next
    while(head != null){
        next = head.next;//先备份
        head.next = pre;//先更新反转后的head.next
        pre = head;//用pre记住前一个位置
        head = next;
    }
    return pre;//pre现在是最后一个节点
}

//反转双链表
public static DoubleListNode reverseDoubleList(DoubleListNode head){
    //先准备两个指针
    DoubleListNode pre = null;
    DoubleListNode next = null;
    //需要更新的元素：head, head.next, head.pre
    while(head != null){
        next = head.next;//先备份
        head.next = head.last;//先修改备份的元素
        head.last = next;
        pre = head;
        head = next;
    }
    return pre;
}
