//随机快速排序经典版（不推荐）
public static void quickSort1(int l, int r){
    if(l >= r) return;
    //随机这一下，常数时间比较大
    //但只有这一下随机才能在概率上把快速排序的时间复杂度收敛到O(nlogn)
    int x = arr[l+(int)(Math.random()*(r-l+1))];
    int mid = partition1(l, r, x);
    quickSort1(l, mid-1);
    quickSort1(mid+1, r);
}

//已知arr[l...r]范围上一定有x这个值
//划分数组<=x放左边，>x放右边，
// 并确保划分完成时左侧区域最后一个数字是x
public static int partition1(int l, int r, int x){
    //a: arr[l...a-1]是左侧区域
    //xi: 记录左侧区域任何一个x的位置，哪一个都可以
    int a = l, xi = 0;
    for(int i = l; i <= r; i++){
        if(arr[i] <= arr[a]){
            swap(a,i);
            if(arr[a] == x){
                xi = a;
            }
            a++;
        }
    }
    swap(xi, a-1);
    return a-1;
}

//随机快速排序改进版（推荐）
public static void quickSort2(int l, int r){
    if(l >= r) return;
    int x = arr[l+(int)Math.random()*(r-l+1)];
    partition2(l,r,x);
    //左部分的递归会改变last值，在每个函数里都添加标记
    //使用临时变量left,right
    int left = first;
    int right = last;
    quickSort2(l, left-1);
    quickSort2(right+1,r);

}
//荷兰国旗问题
//一次性解决所有==x的数
//设置三个分区<,=,>x，两个变量first,last维护两边区域，指针i往右走
public static int first, last;
public static void partition2(int l, int r, int x){
    first = l, last = r;
    int i = l;
    while(i<=last){
        if(arr[i] == x) i++;
        else if(arr[i] < x) swap(first++, i++);
        else swap(i, last--);//注意如果>x时i不++，
        // 还要再次比较从last换过来的数
    }
}

public static void swap(int a, int b){
    int tmp = arr[a];
    arr[a] = arr[b];
    arr[b] = tmp;
}
